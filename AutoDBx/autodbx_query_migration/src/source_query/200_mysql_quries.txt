-- Query 151: Complex aggregation with JSON
SELECT u.username, 
       COUNT(o.order_id) as total_orders,
       SUM(o.total_amount) as total_spent,
       JSON_EXTRACT(u.profile_json, '$.preferences.notifications') as notifications
FROM users u
LEFT JOIN orders o ON u.user_id = o.user_id
LEFT JOIN order_items oi ON o.order_id = oi.order_id
LEFT JOIN products p ON oi.product_id = p.product_id
WHERE DATE_FORMAT(u.created_date, '%Y-%m') = DATE_FORMAT(NOW(), '%Y-%m')
GROUP BY u.user_id, u.username, u.profile_json
HAVING SUM(o.total_amount) > (
    SELECT AVG(total_amount) 
    FROM orders 
    WHERE YEAR(order_date) = YEAR(NOW())
);

-- Query 152: CTE with multiple joins
WITH monthly_sales AS (
    SELECT DATE_FORMAT(o.order_date, '%Y-%m') as month,
           COUNT(*) as order_count,
           SUM(o.total_amount) as revenue
    FROM orders o
    WHERE o.order_date >= DATE_SUB(NOW(), INTERVAL 12 MONTH)
    GROUP BY DATE_FORMAT(o.order_date, '%Y-%m')
),
product_performance AS (
    SELECT p.product_id, 
           p.name,
           COUNT(oi.item_id) as times_ordered,
           JSON_EXTRACT(p.attributes_json, '$.category') as json_category
    FROM products p
    JOIN order_items oi ON p.product_id = oi.product_id
    JOIN orders o ON oi.order_id = o.order_id
    WHERE DATEDIFF(NOW(), o.order_date) <= 90
    GROUP BY p.product_id, p.name, p.attributes_json
)
SELECT ms.month, ms.revenue, pp.name as top_product
FROM monthly_sales ms
CROSS JOIN product_performance pp
WHERE pp.times_ordered = (
    SELECT MAX(times_ordered) 
    FROM product_performance
);

WITH product_features AS (
    SELECT p.product_id,
           p.name,
           p.price,
           JSON_EXTRACT(p.attributes_json, '$.features') as features_json,
           JSON_LENGTH(JSON_EXTRACT(p.attributes_json, '$.features')) as feature_count

    SELECT pf.product_id,
           pf.name,
           pf.price,
           pf.feature_count,
           JSON_EXTRACT(pf.features_json, CONCAT('$[', numbers.n, ']')) as individual_feature
    FROM product_features pf
    CROSS JOIN (
        SELECT 0 as n UNION SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 
        UNION SELECT 5 UNION SELECT 6 UNION SELECT 7 UNION SELECT 8 UNION SELECT 9
    ) numbers
    WHERE numbers.n < pf.feature_count
),

    SELECT individual_feature,
           COUNT(*) as products_with_feature,
           AVG(price) as avg_price_with_feature,
           MIN(price) as min_price,

    WHERE individual_feature IS NOT NULL

    HAVING COUNT(*) >= 2
),
product_sales AS (
    SELECT p.product_id,

-- Query 354: Time series analysis with moving averages
WITH daily_metrics AS (
    SELECT DATE(o.order_date) as order_date,
           COUNT(*) as daily_orders,
           SUM(o.total_amount) as daily_revenue,
           COUNT(DISTINCT o.user_id) as daily_customers,
           AVG(o.total_amount) as daily_avg_order_value
    FROM orders o
    WHERE o.order_date >= DATE_SUB(NOW(), INTERVAL 90 DAY)
    GROUP BY DATE(o.order_date)
),
moving_averages AS (
    SELECT order_date,
           daily_orders,
           daily_revenue,
           daily_customers,
           daily_avg_order_value,
           AVG(daily_orders) OVER (
               ORDER BY order_date 
               ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
           ) as orders_7day_ma,
           AVG(daily_revenue) OVER (
               ORDER BY order_date 
               ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
           ) as revenue_7day_ma,
           AVG(daily_customers) OVER (
               ORDER BY order_date 
               ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
           ) as customers_7day_ma,
           LAG(daily_revenue, 7) OVER (ORDER BY order_date) as revenue_week_ago,
           LAG(daily_revenue, 30) OVER (ORDER BY order_date) as revenue_month_ago
    FROM daily_metrics
)
SELECT order_date,
       daily_orders,
       daily_revenue,
       orders_7day_ma,
       revenue_7day_ma,
       customers_7day_ma,
       CASE 
           WHEN revenue_week_ago > 0 
           THEN ROUND((daily_revenue - revenue_week_ago) * 100.0 / revenue_week_ago, 2)
           ELSE NULL 
       END as week_over_week_growth,
       CASE 
           WHEN revenue_month_ago > 0 
           THEN ROUND((daily_revenue - revenue_month_ago) * 100.0 / revenue_month_ago, 2)
           ELSE NULL 
       END as month_over_month_growth,
       CASE 
           WHEN daily_revenue > revenue_7day_ma * 1.2 THEN 'High'
           WHEN daily_revenue < revenue_7day_ma * 0.8 THEN 'Low'
           ELSE 'Normal'
       END as revenue_anomaly
FROM moving_averages
ORDER BY order_date DESC;

    LEFT JOIN orders o ON u.user_id = o.user_id 
        AND o.order_date >= DATE_SUB(NOW(), INTERVAL 18 MONTH)
    LEFT JOIN order_items oi ON o.order_id = oi.order_id
    LEFT JOIN products p ON oi.product_id = p.product_id
    WHERE u.created_date >= DATE_SUB(NOW(), INTERVAL 3 YEAR)
    GROUP BY u.user_id, u.username, u.profile_json
)
SELECT ubm.username,
       ubm.user_segment,
       ubm.order_frequency,
       ubm.avg_order_value,
       ubm.lifetime_value,
       ubm.category_diversity,
       ct.path as favorite_category_path,
       PERCENT_RANK() OVER (ORDER BY ubm.lifetime_value) as ltv_percentile,
       NTILE(10) OVER (ORDER BY ubm.order_frequency) as frequency_decile,
       CASE 
           WHEN ubm.order_frequency >= 20 AND ubm.avg_order_value >= 200 THEN 'VIP'
           WHEN ubm.order_frequency >= 10 AND ubm.category_diversity >= 3 THEN 'Power User'
           WHEN ubm.lifetime_value >= 1000 THEN 'High Value'
           WHEN ubm.active_months >= 12 THEN 'Loyal'
           ELSE 'Regular'
       END as customer_tier,
       JSON_LENGTH(ubm.purchased_categories) as unique_categories_purchased,
       ROUND(ubm.order_value_variance, 2) as spending_consistency
FROM user_behavior_matrix ubm
LEFT JOIN category_tree ct ON JSON_CONTAINS(ubm.purchased_categories, JSON_QUOTE(ct.name))
WHERE ubm.order_frequency > 0
ORDER BY ubm.lifetime_value DESC, ubm.order_frequency DESC
LIMIT 100;

-- Query 356: Advanced time-based customer segmentation
SELECT user_segment,
       COUNT(*) as segment_size,
       AVG(order_frequency) as avg_orders,
       AVG(lifetime_value) as avg_ltv,
       AVG(category_diversity) as avg_category_diversity,
       PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY lifetime_value) as median_ltv,
       PERCENTILE_CONT(0.9) WITHIN GROUP (ORDER BY lifetime_value) as p90_ltv
FROM (
    SELECT u.user_id,
           CASE 
               WHEN DATEDIFF(NOW(), MAX(o.order_date)) <= 30 THEN 'Active'
               WHEN DATEDIFF(NOW(), MAX(o.order_date)) <= 90 THEN 'At Risk'
               WHEN DATEDIFF(NOW(), MAX(o.order_date)) <= 180 THEN 'Dormant'
               ELSE 'Lost'
           END as user_segment,
           COUNT(o.order_id) as order_frequency,
           SUM(o.total_amount) as lifetime_value,
           COUNT(DISTINCT p.category) as category_diversity
    FROM users u
    LEFT JOIN orders o ON u.user_id = o.user_id
    LEFT JOIN order_items oi ON o.order_id = oi.order_id
    LEFT JOIN products p ON oi.product_id = p.product_id
    WHERE u.created_date >= DATE_SUB(NOW(), INTERVAL 2 YEAR)
    GROUP BY u.user_id
) segmented_users
GROUP BY user_segment;

-- Query 357: Product recommendation engine data
WITH user_product_affinity AS (
    SELECT u.user_id,
           p.product_id,
           COUNT(*) as purchase_count,
           SUM(oi.quantity) as total_quantity,
           AVG(r.rating) as avg_rating,
           MAX(o.order_date) as last_purchase_date,
           JSON_EXTRACT(p.attributes_json, '$.category') as product_category,
           JSON_EXTRACT(p.attributes_json, '$.subcategory') as product_subcategory
    FROM users u
    JOIN orders o ON u.user_id = o.user_id
    JOIN order_items oi ON o.order_id = oi.order_id
    JOIN products p ON oi.product_id = p.product_id
    LEFT JOIN reviews r ON u.user_id = r.user_id AND p.product_id = r.product_id
    WHERE o.order_date >= DATE_SUB(NOW(), INTERVAL 1 YEAR)
    GROUP BY u.user_id, p.product_id, p.attributes_json
),
product_similarity AS (
    SELECT p1.product_id as product_a,
           p2.product_id as product_b,
           COUNT(DISTINCT upa1.user_id) as common_buyers,
           AVG(upa1.avg_rating) as avg_rating_a,
           AVG(upa2.avg_rating) as avg_rating_b,
           CASE 
               WHEN JSON_EXTRACT(p1.attributes_json, '$.category') = JSON_EXTRACT(p2.attributes_json, '$.category') 
               THEN 'Same Category'
               WHEN JSON_EXTRACT(p1.attributes_json, '$.subcategory') = JSON_EXTRACT(p2.attributes_json, '$.subcategory')
               THEN 'Same Subcategory'
               ELSE 'Different'
           END as category_relationship
    FROM user_product_affinity upa1
    JOIN user_product_affinity upa2 ON upa1.user_id = upa2.user_id AND upa1.product_id < upa2.product_id
    JOIN products p1 ON upa1.product_id = p1.product_id
    JOIN products p2 ON upa2.product_id = p2.product_id
    GROUP BY p1.product_id, p2.product_id, p1.attributes_json, p2.attributes_json
    HAVING COUNT(DISTINCT upa1.user_id) >= 3
)
SELECT ps.product_a,
       ps.product_b,
       ps.common_buyers,
       ps.category_relationship,
       ROUND((ps.avg_rating_a + ps.avg_rating_b) / 2, 2) as combined_rating,
       RANK() OVER (PARTITION BY ps.product_a ORDER BY ps.common_buyers DESC) as similarity_rank
FROM product_similarity ps
ORDER BY ps.product_a, ps.common_buyers DESC;

-- Query 358: Seasonal demand forecasting
WITH monthly_product_sales AS (
    SELECT p.product_id,
           p.name,
           DATE_FORMAT(o.order_date, '%Y-%m') as sales_month,
           MONTH(o.order_date) as month_number,
           YEAR(o.order_date) as year_number,
           SUM(oi.quantity) as units_sold,
           SUM(oi.quantity * oi.unit_price) as revenue,
           COUNT(DISTINCT o.order_id) as order_count,
           JSON_EXTRACT(p.attributes_json, '$.seasonal_category') as seasonal_category
    FROM products p
    JOIN order_items oi ON p.product_id = oi.product_id
    JOIN orders o ON oi.order_id = o.order_id
    WHERE o.order_date >= DATE_SUB(NOW(), INTERVAL 24 MONTH)
    GROUP BY p.product_id, p.name, DATE_FORMAT(o.order_date, '%Y-%m'), 
             MONTH(o.order_date), YEAR(o.order_date), p.attributes_json
),
seasonal_patterns AS (
    SELECT product_id,
           name,
           month_number,
           seasonal_category,
           AVG(units_sold) as avg_monthly_units,
           STDDEV(units_sold) as units_volatility,
           AVG(revenue) as avg_monthly_revenue,
           COUNT(*) as data_points
    FROM monthly_product_sales
    GROUP BY product_id, name, month_number, seasonal_category
    HAVING COUNT(*) >= 2
),
trend_analysis AS (
    SELECT sp.product_id,
           sp.name,
           sp.seasonal_category,
           MAX(CASE WHEN sp.month_number = 12 THEN sp.avg_monthly_units END) as december_avg,
           MAX(CASE WHEN sp.month_number = 6 THEN sp.avg_monthly_units END) as june_avg,
           AVG(sp.avg_monthly_units) as overall_avg,
           MAX(sp.avg_monthly_units) as peak_month_units,
           MIN(sp.avg_monthly_units) as low_month_units,
           (MAX(sp.avg_monthly_units) - MIN(sp.avg_monthly_units)) / AVG(sp.avg_monthly_units) as seasonality_index
    FROM seasonal_patterns sp
    GROUP BY sp.product_id, sp.name, sp.seasonal_category
)
SELECT ta.name,
       ta.seasonal_category,
       ta.overall_avg,
       ta.peak_month_units,
       ta.low_month_units,
       ROUND(ta.seasonality_index, 3) as seasonality_score,
       CASE 
           WHEN ta.seasonality_index > 1.0 THEN 'High Seasonal'
           WHEN ta.seasonality_index > 0.5 THEN 'Moderate Seasonal'
           ELSE 'Low Seasonal'
       END as seasonality_category,
       ROUND(ta.december_avg / NULLIF(ta.june_avg, 0), 2) as winter_summer_ratio
FROM trend_analysis ta
WHERE ta.overall_avg > 0
ORDER BY ta.seasonality_index DESC;

-- Query 359: Advanced inventory optimization
WITH product_velocity AS (
    SELECT p.product_id,
           p.name,
           p.price,
           JSON_EXTRACT(p.attributes_json, '$.cost') as product_cost,
           JSON_EXTRACT(p.attributes_json, '$.weight') as product_weight,
           COUNT(DISTINCT o.order_id) as order_frequency,
           SUM(oi.quantity) as total_units_sold,
           AVG(oi.quantity) as avg_units_per_order,
           DATEDIFF(NOW(), MIN(o.order_date)) as days_in_catalog,
           DATEDIFF(NOW(), MAX(o.order_date)) as days_since_last_sale
    FROM products p
    LEFT JOIN order_items oi ON p.product_id = oi.product_id
    LEFT JOIN orders o ON oi.order_id = o.order_id
        AND o.order_date >= DATE_SUB(NOW(), INTERVAL 12 MONTH)
    WHERE p.created_date >= DATE_SUB(NOW(), INTERVAL 18 MONTH)
    GROUP BY p.product_id, p.name, p.price, p.attributes_json
),
inventory_metrics AS (
    SELECT pv.*,
           CASE 
               WHEN pv.days_in_catalog > 0 
               THEN pv.total_units_sold / (pv.days_in_catalog / 30.0)
               ELSE 0 
           END as monthly_velocity,
           CASE 
               WHEN pv.product_cost IS NOT NULL AND pv.product_cost > 0
               THEN (pv.price - CAST(pv.product_cost AS DECIMAL(10,2))) / pv.price
               ELSE NULL
           END as profit_margin,
           CASE 
               WHEN pv.days_since_last_sale <= 7 THEN 'Hot'
               WHEN pv.days_since_last_sale <= 30 THEN 'Warm'
               WHEN pv.days_since_last_sale <= 90 THEN 'Cool'
               ELSE 'Cold'
           END as demand_temperature
    FROM product_velocity pv
),
abc_analysis AS (
    SELECT *,
           NTILE(3) OVER (ORDER BY total_units_sold DESC) as volume_tier,
           NTILE(3) OVER (ORDER BY monthly_velocity DESC) as velocity_tier,
           NTILE(3) OVER (ORDER BY profit_margin DESC) as margin_tier
    FROM inventory_metrics
    WHERE total_units_sold > 0
)
SELECT name,
       price,
       total_units_sold,
       monthly_velocity,
       profit_margin,
       demand_temperature,
       CASE 
           WHEN volume_tier = 1 AND velocity_tier = 1 THEN 'A - High Volume, High Velocity'
           WHEN volume_tier = 1 OR velocity_tier = 1 THEN 'B - Medium Priority'
           ELSE 'C - Low Priority'
       END as abc_category,
       CASE 
           WHEN demand_temperature = 'Cold' AND days_since_last_sale > 180 THEN 'Consider Discontinuing'
           WHEN monthly_velocity < 1 AND profit_margin < 0.2 THEN 'Review Pricing'
           WHEN volume_tier = 1 AND velocity_tier = 1 THEN 'Increase Stock'
           ELSE 'Monitor'
       END as recommendation
FROM abc_analysis
ORDER BY monthly_velocity DESC, total_units_sold DESC;

-- Query 360: Customer churn prediction features
WITH customer_features AS (
    SELECT u.user_id,
           u.username,
           DATEDIFF(NOW(), u.created_date) as tenure_days,
           COUNT(DISTINCT o.order_id) as total_orders,
           SUM(o.total_amount) as total_spent,
           AVG(o.total_amount) as avg_order_value,
           DATEDIFF(NOW(), MAX(o.order_date)) as days_since_last_order,
           COUNT(DISTINCT DATE_FORMAT(o.order_date, '%Y-%m')) as active_months,
           COUNT(DISTINCT p.category) as categories_purchased,
           AVG(r.rating) as avg_rating_given,
           COUNT(r.review_id) as reviews_written,
           STDDEV(o.total_amount) as order_value_variance,
           JSON_EXTRACT(u.profile_json, '$.support_tickets') as support_interactions,
           JSON_EXTRACT(u.profile_json, '$.email_engagement') as email_engagement
    FROM users u
    LEFT JOIN orders o ON u.user_id = o.user_id
    LEFT JOIN order_items oi ON o.order_id = oi.order_id
    LEFT JOIN products p ON oi.product_id = p.product_id
    LEFT JOIN reviews r ON u.user_id = r.user_id
    WHERE u.created_date <= DATE_SUB(NOW(), INTERVAL 90 DAY)
    GROUP BY u.user_id, u.username, u.created_date, u.profile_json
),
churn_indicators AS (
    SELECT *,
           CASE 
               WHEN days_since_last_order > 180 THEN 1 
               ELSE 0 
           END as likely_churned,
           CASE 
               WHEN total_orders > 0 THEN total_spent / total_orders 
               ELSE 0 
           END as calculated_aov,
           CASE 
               WHEN tenure_days > 0 THEN total_orders / (tenure_days / 30.0)
               ELSE 0 
           END as orders_per_month,
           CASE 
               WHEN active_months > 0 AND tenure_days > 0 
               THEN active_months / (tenure_days / 30.0)
               ELSE 0 
           END as activity_ratio
    FROM customer_features
)
SELECT username,
       tenure_days,
       total_orders,
       total_spent,
       days_since_last_order,
       orders_per_month,
       activity_ratio,
       categories_purchased,
       avg_rating_given,
       likely_churned,
       CASE 
           WHEN likely_churned = 1 THEN 'Churned'
           WHEN days_since_last_order > 90 THEN 'High Risk'
           WHEN days_since_last_order > 60 THEN 'Medium Risk'
           WHEN orders_per_month < 0.5 THEN 'Low Engagement'
           ELSE 'Active'
       END as churn_risk_segment,
       NTILE(10) OVER (ORDER BY total_spent DESC) as value_decile,
       NTILE(10) OVER (ORDER BY orders_per_month DESC) as frequency_decile
FROM churn_indicators
WHERE total_orders > 0
ORDER BY days_since_last_order DESC, total_spent DESC;

-- Query 361: Cross-sell opportunity analysis
WITH purchase_combinations AS (
    SELECT o.order_id,
           GROUP_CONCAT(DISTINCT p.category ORDER BY p.category) as category_combo,
           COUNT(DISTINCT p.category) as categories_in_order,
           SUM(oi.quantity * oi.unit_price) as order_value
    FROM orders o
    JOIN order_items oi ON o.order_id = oi.order_id
    JOIN products p ON oi.product_id = p.product_id
    WHERE o.order_date >= DATE_SUB(NOW(), INTERVAL 6 MONTH)
    GROUP BY o.order_id
),
combination_analysis AS (
    SELECT category_combo,
           categories_in_order,
           COUNT(*) as combo_frequency,
           AVG(order_value) as avg_combo_value,
           SUM(order_value) as total_combo_value
    FROM purchase_combinations
    WHERE categories_in_order >= 2
    GROUP BY category_combo, categories_in_order
    HAVING COUNT(*) >= 5
)
SELECT category_combo,
       categories_in_order,
       combo_frequency,
       avg_combo_value,
       total_combo_value,
       RANK() OVER (ORDER BY combo_frequency DESC) as popularity_rank,
       RANK() OVER (ORDER BY avg_combo_value DESC) as value_rank
FROM combination_analysis
ORDER BY combo_frequency DESC;
